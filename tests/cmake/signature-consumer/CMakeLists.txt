cmake_minimum_required(VERSION 3.20)
project(QuietProofSignatureConsumer LANGUAGES CXX)
if(WITH_RPC)
  find_package(LongfellowZKQuietProof CONFIG REQUIRED COMPONENTS signature rpc)
else()
  find_package(LongfellowZKQuietProof CONFIG REQUIRED COMPONENTS signature)
  if(TARGET CURL::libcurl OR TARGET LongfellowZKQuietProof::rpc)
    message(FATAL_ERROR "Signature-only lookup loaded RPC")
  endif()
endif()
if(TARGET LongfellowZKECDSA::ecdsa OR TARGET libsecp256k1::secp256k1)
  message(FATAL_ERROR "Signature lookup loaded an unreviewed ECDSA or oracle dependency")
endif()
get_target_property(signature_links LongfellowZKQuietProof::signature INTERFACE_LINK_LIBRARIES)
if(NOT signature_links STREQUAL "LongfellowZKQuietProof::quietproof")
  message(FATAL_ERROR "Signature export must depend only on the base QuietProof target")
endif()
add_executable(signature-consumer main.cc)
target_link_libraries(signature-consumer PRIVATE LongfellowZKQuietProof::signature)
if(WITH_RPC)
  target_link_libraries(signature-consumer PRIVATE LongfellowZKQuietProof::rpc)
endif()
