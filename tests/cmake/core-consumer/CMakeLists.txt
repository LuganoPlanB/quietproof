cmake_minimum_required(VERSION 3.20)
project(QuietProofCoreConsumer LANGUAGES CXX)
find_package(LongfellowZKQuietProof CONFIG REQUIRED)
if(TARGET CURL::libcurl OR TARGET LongfellowZKQuietProof::rpc)
  message(FATAL_ERROR "Core-only package lookup loaded the optional RPC dependency")
endif()
get_target_property(core_links LongfellowZKQuietProof::quietproof INTERFACE_LINK_LIBRARIES)
if(core_links MATCHES "[Cc][Uu][Rr][Ll]|::rpc")
  message(FATAL_ERROR "Core interface contains a network dependency")
endif()
add_executable(core-consumer main.cc)
target_link_libraries(core-consumer PRIVATE LongfellowZKQuietProof::quietproof)
