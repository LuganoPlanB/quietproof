cmake_minimum_required(VERSION 3.20)
project(QuietProofRpcConsumer LANGUAGES CXX)
find_package(LongfellowZKQuietProof CONFIG REQUIRED COMPONENTS rpc)
if(EXPECT_STATIC_CURL)
  get_target_property(curl_location CURL::libcurl IMPORTED_LOCATION_RELEASE)
  if(NOT curl_location)
    get_target_property(curl_location CURL::libcurl IMPORTED_LOCATION)
  endif()
  if(NOT curl_location MATCHES "\\.a$")
    message(FATAL_ERROR "Static curl matrix selected a non-archive: ${curl_location}")
  endif()
endif()
get_target_property(rpc_links LongfellowZKQuietProof::rpc INTERFACE_LINK_LIBRARIES)
if(NOT rpc_links MATCHES "LINK_ONLY:CURL::libcurl")
  message(FATAL_ERROR "Static RPC export lost its link-only curl dependency")
endif()
# A shared consumer also requires the installed static archive to be PIC.
add_library(rpc-consumer-library SHARED consumer.cc)
target_link_libraries(rpc-consumer-library PRIVATE LongfellowZKQuietProof::rpc)
add_executable(rpc-consumer main.cc)
target_link_libraries(rpc-consumer PRIVATE rpc-consumer-library)
